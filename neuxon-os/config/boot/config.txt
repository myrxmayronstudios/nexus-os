# Neuxon OS - Raspberry Pi 3B+ Boot Configuration
# BCM2837B0 - ARM Cortex-A53 aarch64

# Force 64-bit kernel mode
arm_64bit=1

# Boot optimization
boot_delay=0
disable_splash=1
force_turbo=0

# GPU & Display - VideoCore IV KMS Driver with 128MB CMA cap to prevent 1GB RAM pool starvation
dtoverlay=vc4-kms-v3d,cma-128
gpu_mem=64
max_framebuffers=2
disable_overscan=1
hdmi_force_hotplug=1

# Hardware peripherals
dtparam=audio=on
dtparam=i2c_arm=off
dtparam=i2s=off
dtparam=spi=off

# Disable Bluetooth to save memory and release PL011 UART
dtoverlay=disable-bt

# Serial Console
enable_uart=1
